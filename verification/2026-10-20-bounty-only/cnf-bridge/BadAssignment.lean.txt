-- Deliberate defect: auxiliary columns have threshold j+2 instead of j+1.
def badAssignment (B : Finset ℕ) : Sat.Valuation := fun i =>
  if i < 12 then i + 1 ∈ B
  else if i < 120 then aux (fun k => k + 1 ∈ B) ((i - 12) / 9 + 1) ((i - 12) % 9 + 1)
  else if i < 156 then aux (fun k => k + 1 ∉ B) ((i - 120) / 3 + 1) ((i - 120) % 3 + 1)
  else False
-- Archived clause 11 is [-1,13]. The valid proof must reject this wrong assignment.
example (B : Finset ℕ) :
    (badAssignment B).satisfies [Sat.Literal.neg 0, Sat.Literal.pos 12] := by
  change (1 ∈ B) → (¬ aux (fun k => k + 1 ∈ B) 1 0) → False
  intro hx hn
  exact hn (aux_of_true (fun k => k + 1 ∈ B) hx)
