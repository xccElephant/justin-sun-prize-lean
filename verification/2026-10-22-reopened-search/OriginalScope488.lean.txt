lemma multiples_of_one_mem {A : Finset ℕ} (h : 1 ∈ A) (x : ℕ) :
    multiples A x = Finset.Icc 1 x := by
  ext j
  simp only [multiples, Finset.mem_filter]
  constructor
  · exact And.left
  · intro hj
    exact ⟨hj, 1, h, one_dvd j⟩

/-- The prior counterexample respects both excluded generators 0 and 1,
and violates the original rational-density bound strictly. -/
theorem counterexample_original_scope :
    ∃ A : Finset ℕ, A.Nonempty ∧ 0 ∉ A ∧ 1 ∉ A ∧
      ∃ n m : ℕ, 0 < n ∧ n < m ∧ (∀ a ∈ A, a ≤ n) ∧
      2 * ((multiples A n).card : ℚ) / n < ((multiples A m).card : ℚ) / m := by
  obtain ⟨A,hA,h0,n,m,hmax,hnm,hbad⟩ := finite_counterexample
  have hn : 0 < n := by
    obtain ⟨a,ha⟩ := hA
    have ha0 : a ≠ 0 := by intro h; exact h0 (h ▸ ha)
    exact lt_of_lt_of_le (Nat.pos_of_ne_zero ha0) (hmax a ha)
  have hm : 0 < m := hn.trans hnm
  have h1 : 1 ∉ A := by
    intro hone
    rw [multiples_of_one_mem hone, multiples_of_one_mem hone] at hbad
    simp only [Nat.card_Icc, Nat.add_sub_cancel] at hbad
    nlinarith
  refine ⟨A,hA,h0,h1,n,m,hn,hnm,hmax,?_⟩
  apply (div_lt_div_iff₀ (by exact_mod_cast hn : (0 : ℚ) < n)
    (by exact_mod_cast hm : (0 : ℚ) < m)).mpr
  have hbadQ : (2 : ℚ) * m * (multiples A n).card < n * (multiples A m).card := by
    exact_mod_cast hbad
  nlinarith

/-- Literal FormalConjectures formulation including rational division,
positive counts, both excluded generators and the Finset.max premise. -/
theorem original_statement_false : ¬ (∀ A : Finset ℕ, A.Nonempty →
    0 ∉ A → 1 ∉ A →
    let B := {j : ℕ | 1 ≤ j ∧ ∃ a ∈ A, a ∣ j}
    ∀ (n m : ℕ), m > n → A.max ≤ n →
      ((Finset.Icc 1 m).filter (· ∈ B)).card / (m : ℚ) <
        2 * ((Finset.Icc 1 n).filter (· ∈ B)).card / (n : ℚ)) := by
  intro h
  obtain ⟨A,hA,h0,h1,n,m,hn,hnm,hmax,hbad⟩ := counterexample_original_scope
  have hmax' : A.max ≤ (n : WithBot ℕ) := Finset.max_le_iff.mpr (by
    intro a ha
    exact WithBot.coe_le_coe.mpr (hmax a ha))
  have hgood := h A hA h0 h1 n m hnm hmax'
  have heq (x : ℕ) :
      (Finset.Icc 1 x).filter (fun j => 1 ≤ j ∧ ∃ a ∈ A, a ∣ j) = multiples A x := by
    ext j
    simp only [Finset.mem_filter, Finset.mem_Icc, multiples]
    tauto
  have hgood' : ((multiples A m).card : ℚ) / m <
      2 * ((multiples A n).card : ℚ) / n := by
    simpa only [Set.mem_ofPred_eq, heq] using hgood
  exact (not_lt_of_gt hbad) hgood'
